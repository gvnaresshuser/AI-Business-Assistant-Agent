Generate payment records
Now let's populate sales_payments_biz_agent.

Our rules:
- Delivered orders → Completed payment.
- Shipped orders → Completed payment.
- Processing orders → Pending payment.
- Pending orders → Pending payment.
- Cancelled orders → No payment record.
- Payment amount → Matches the order total.

    COUNT(*) AS total_non_cancelled_orders,

    ON p.business_id = o.business_id
   AND p.order_id = o.id

CREATE UNIQUE INDEX IF NOT EXISTS
uq_payments_business_id_id_biz_agent
ON sales_payments_biz_agent(business_id, id);

CREATE TABLE IF NOT EXISTS sales_refunds_biz_agent (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    business_id UUID NOT NULL,
    order_id UUID NOT NULL,
    payment_id UUID,

    refund_reference VARCHAR(100),
    refund_date TIMESTAMPTZ DEFAULT NOW(),
    refund_reason TEXT,

    refund_status VARCHAR(30) DEFAULT 'Pending',
    amount NUMERIC(12,2) NOT NULL,

    CHECK (amount > 0),

    CHECK (
        refund_status IN ('Pending', 'Completed', 'Failed')
    ),

    FOREIGN KEY (business_id, order_id)
        REFERENCES sales_orders_biz_agent(business_id, id),

    FOREIGN KEY (business_id, payment_id)
        REFERENCES sales_payments_biz_agent(business_id, id)
);

CREATE INDEX IF NOT EXISTS idx_refunds_order_biz_agent
ON sales_refunds_biz_agent(business_id, order_id);

CREATE INDEX IF NOT EXISTS idx_refunds_date_biz_agent
ON sales_refunds_biz_agent(business_id, refund_date);

CREATE INDEX IF NOT EXISTS idx_refunds_status_biz_agent
ON sales_refunds_biz_agent(business_id, refund_status);


SELECT
    table_name
FROM information_schema.tables
WHERE table_schema = 'public'
AND table_name = 'sales_refunds_biz_agent';

Full vs partial refunds

SELECT
    CASE
        WHEN r.amount = o.total_amount THEN 'Full Refund'
        ELSE 'Partial Refund'
    END AS refund_type,
    COUNT(*) AS total_records,
    ROUND(SUM(r.amount), 2) AS refund_amount
FROM sales_refunds_biz_agent r
JOIN sales_orders_biz_agent o
    ON o.business_id = r.business_id
   AND o.id = r.order_id

BEGIN;

-- 1. Purchase receipt movements
-- Use received purchase quantities as inbound stock movements.

INSERT INTO inv_stock_movements_biz_agent
(
    business_id,
    product_id,
    movement_type,
    quantity,
    reference_type,
    reference_id,
    movement_date,
    notes
)
SELECT
    pi.business_id,
    pi.product_id,
    'PURCHASE_RECEIPT',
    pi.received_quantity,
    'PURCHASE_ORDER',
    pi.purchase_order_id,
    po.order_date,
    'Historical purchase receipt - demo data'
FROM purch_purchase_items_biz_agent pi
JOIN purch_purchase_orders_biz_agent po
    ON po.business_id = pi.business_id
   AND po.id = pi.purchase_order_id
WHERE pi.received_quantity > 0
  AND po.status IN ('Received', 'Partially Received')
  AND NOT EXISTS (
      SELECT 1
      FROM inv_stock_movements_biz_agent sm
      WHERE sm.reference_type = 'PURCHASE_ORDER'
        AND sm.reference_id = pi.purchase_order_id
        AND sm.product_id = pi.product_id
        AND sm.movement_type = 'PURCHASE_RECEIPT'
  );

-- 2. Sales movements
-- Select 150 delivered orders and create one movement
-- for each order item in those orders.

WITH selected_orders AS (
    SELECT
        id,
        business_id
    FROM sales_orders_biz_agent
    WHERE order_status = 'Delivered'
    ORDER BY order_date DESC, id
    LIMIT 150
)
INSERT INTO inv_stock_movements_biz_agent
(
    business_id,
    product_id,
    movement_type,
    quantity,
    reference_type,
    reference_id,
    movement_date,
    notes
)
SELECT
    oi.business_id,
    oi.product_id,
    'SALE',
    -oi.quantity,
    'SALES_ORDER',
    oi.order_id,
    so.order_date,
    'Historical delivered sale - demo data'
FROM sales_order_items_biz_agent oi
JOIN selected_orders selected
    ON selected.business_id = oi.business_id
   AND selected.id = oi.order_id
JOIN sales_orders_biz_agent so
    ON so.business_id = oi.business_id
   AND so.id = oi.order_id
WHERE NOT EXISTS (
    SELECT 1
    FROM inv_stock_movements_biz_agent sm
    WHERE sm.reference_type = 'SALES_ORDER'
      AND sm.reference_id = oi.order_id
      AND sm.product_id = oi.product_id
      AND sm.movement_type = 'SALE'
);

-- 3. Customer return movements
-- Use 20 completed refund records as demonstration references.

WITH selected_refunds AS (
    SELECT
        id,
        business_id,
        order_id,
        refund_date
    FROM sales_refunds_biz_agent
    WHERE status = 'Completed'
    ORDER BY refund_date DESC NULLS LAST, id
    LIMIT 20
)
INSERT INTO inv_stock_movements_biz_agent
(
    business_id,
    product_id,
    movement_type,
    quantity,
    reference_type,
    reference_id,
    movement_date,
    notes
)
SELECT
    oi.business_id,
    oi.product_id,
    'CUSTOMER_RETURN',
    1,
    'SALES_ORDER',
    sr.order_id,
    COALESCE(sr.refund_date, NOW()),
    'Sample customer return associated with completed refund'
FROM selected_refunds sr
JOIN sales_order_items_biz_agent oi
    ON oi.business_id = sr.business_id
   AND oi.order_id = sr.order_id
WHERE NOT EXISTS (
    SELECT 1
    FROM inv_stock_movements_biz_agent sm
    WHERE sm.reference_type = 'SALES_ORDER'
      AND sm.reference_id = sr.order_id
      AND sm.product_id = oi.product_id
      AND sm.movement_type = 'CUSTOMER_RETURN'
);

-- 4. Stock adjustment movements
-- Create 10 sample adjustments.

INSERT INTO inv_stock_movements_biz_agent
(
    business_id,
    product_id,
    movement_type,
    quantity,
    reference_type,
    movement_date,
    notes
)
SELECT
    p.business_id,
    p.id,
    'ADJUSTMENT',
    CASE
        WHEN ROW_NUMBER() OVER (ORDER BY p.sku) % 2 = 0
            THEN -1
        ELSE 1
    END,
    'STOCK_AUDIT',
    NOW() - INTERVAL '5 days',
    'Sample stock audit adjustment - demo data'
FROM prod_products_biz_agent p
WHERE NOT EXISTS (
    SELECT 1
    FROM inv_stock_movements_biz_agent sm
    WHERE sm.product_id = p.id
      AND sm.business_id = p.business_id
      AND sm.reference_type = 'STOCK_AUDIT'
)
ORDER BY p.sku
LIMIT 10;
