SELECT
    business_id,
    document_name,
    COUNT(*) AS total_chunks
FROM agent_knowledge_chunks_biz_agent
GROUP BY
    business_id,
    document_name
ORDER BY
    document_name;
	
	
SELECT
    COUNT(*) AS total_knowledge_chunks,
    COUNT(DISTINCT document_name) AS total_documents
FROM agent_knowledge_chunks_biz_agent;


SELECT
    business_id,
    document_name,
    chunk_index,
    LEFT(chunk_content, 300) AS content_preview,
    metadata
FROM agent_knowledge_chunks_biz_agent
ORDER BY document_name, chunk_index
LIMIT 20;


SELECT
    COUNT(*) AS total_chunks,
    COUNT(embedding) AS chunks_with_embeddings,
    COUNT(*) - COUNT(embedding) AS missing_embeddings
FROM agent_knowledge_chunks_biz_agent;

SELECT
    business_id,
    document_name,
    chunk_index,
    vector_dims(embedding) AS embedding_dimensions,
    embedding::text AS embedding_values
FROM agent_knowledge_chunks_biz_agent
ORDER BY document_name, chunk_index;
